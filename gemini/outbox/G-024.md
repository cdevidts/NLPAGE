## 4. Código (resumen fiel; el Doc trae el completo)
```python
import numpy as np, wave, subprocess
from scipy.signal import butter, sosfilt
sr = 44100
def midi_to_freq(m): return 440.0 * 2 ** ((m - 69) / 12)
def synth_piano_note(freq, dur, velocity=0.8, sr=44100):
    n = int(sr * dur); t = np.linspace(0, dur, n, endpoint=False)
    harmonics = [(1, 1.0), (2, .48), (3, .24), (4, .12), (5, .06), (6, .03)]
    B = 0.00015; sig = np.zeros_like(t)
    for k, amp in harmonics:
        fk = freq * k * np.sqrt(1 + B * k * k)
        if fk < sr / 2:
            d = 2.2 + 0.55 * k
            env = 0.7 * np.exp(-d * t) + 0.3 * np.exp(-0.35 * d * t)
            sig += amp * np.sin(2 * np.pi * fk * t) * env
    a = max(1, int(0.008 * sr)); r = max(1, int(0.025 * sr))
    sig[:a] *= 0.5 * (1 - np.cos(np.pi * np.arange(a) / a))
    if n > r: sig[-r:] *= np.linspace(1, 0, r)
    return sig * velocity
# melodía: lista de (pulso_inicio, midi, pulsos, velocidad) a 0,5 s por pulso;
# acordes: (pulso_inicio, [midi...], pulsos, velocidad); se suman al buffer,
# se normaliza a -1,5 dBFS pico y se guarda con wave (int16).
# ffmpeg -y -i in.wav -codec:a libmp3lame -b:a 192k out.mp3
# Radio: ruido blanco -> FFT * 1/sqrt(f) (rosa) -> butter(4,[300,2800],'bandpass',output='sos')
#  -> *0.22 * (1 + 0.18*sin(2π·6.5·t)); Morse: envolvente por unidades de 80 ms con rampas de 4 ms,
#  tono 700 Hz * 0.62; chasquidos: exp(-180 t)*(sin(2π·1650 t)+0.8·ruido) 45 ms; normalizar a -1 dBFS.
```
